生产篇 01 · 真实认证

把 x-user-id 换掉：每个请求先验身份

x-user-id 谁都能伪造，所以它只能当教学占位。真实认证要回答两个问题：令牌是谁签的（签名）、令牌还给不还给用（过期）。 这里用 node:crypto 实现一个 HMAC 签名的紧凑令牌——JWT 的形、去掉算法协商。它教的是“边界在哪”，生产环境请换成真正的 JWT 库（如 jose）和你的 IdP。

src/auth.ts · 认证模块（完整代码）
import { createHmac, timingSafeEqual } from "node:crypto";
import type { IncomingMessage } from "node:http";

export type AuthUser = {
  userId: string;
  tenantId: string;
};

export class AuthError extends Error {
  status: number;

  constructor(message: string, status: number) {
    super(message);
    this.status = status;
  }
}

// fail-closed：没配密钥就拒绝启动，绝不降级成匿名模式
// （?? "" 是给类型一个非 undefined 的承诺，空值由下一行抛错拦下）
const secret = process.env.AUTH_SECRET ?? "";
if (!secret) {
  throw new Error("必须通过环境变量注入 AUTH_SECRET");
}

function sign(payload: string): string {
  return createHmac("sha256", secret).update(payload).digest("base64url");
}

/** 签发令牌：本地测试或你的登录接口用。生产请换 JWT 库 + IdP。 */
export function issueToken(user: AuthUser, ttlSeconds = 3600): string {
  const payload = JSON.stringify({
    ...user,
    exp: Math.floor(Date.now() / 1000) + ttlSeconds,
  });
  const payloadB64 = Buffer.from(payload).toString("base64url");
  return `${payloadB64}.${sign(payloadB64)}`;
}

/** 每个请求都过这里：验证失败一律 401，不存在“验不过就当访客”。 */
export function authenticate(req: IncomingMessage): AuthUser {
  const header = req.headers["authorization"];
  if (typeof header !== "string" || !header.startsWith("Bearer ")) {
    throw new AuthError("missing bearer token", 401);
  }

  const [payloadB64, signature] = header.slice("Bearer ".length).split(".");
  if (!payloadB64 || !signature) throw new AuthError("malformed token", 401);

  // timingSafeEqual 防时序攻击：比较签名不用 ===
  const expected = Buffer.from(sign(payloadB64));
  const received = Buffer.from(signature);
  if (expected.length !== received.length || !timingSafeEqual(expected, received)) {
    throw new AuthError("bad signature", 401);
  }

  const payload = JSON.parse(Buffer.from(payloadB64, "base64url").toString()) as {
    userId?: string;
    tenantId?: string;
    exp?: number;
  };
  const expired = typeof payload.exp !== "number" || payload.exp * 1000 < Date.now();
  if (!payload.userId || expired) {
    throw new AuthError("token expired or invalid", 401);
  }
  return { userId: payload.userId, tenantId: payload.tenantId ?? "default" };
}

生产篇 02 · 忙时排队

409 是机制教学，排队才是产品行为

实战篇的 409 证明了“同一会话同时只有一个 run”。但真实用户不会接受“请稍后再试”的粗暴拒绝——他们发消息、等回复、再追问。做法是每用户一条串行队列：忙时入队，空闲了按序执行；队列有上限（防止刷爆内存）、有超时（防止排到天荒地老）。

src/queue.ts · 每用户串行队列（完整代码）
type Job = {
  run: () => Promise<void>;
};

export class QueueFullError extends Error {}

export class UserQueue {
  private jobs: Job[] = [];
  private draining = false;
  private limit: number;
  private timeoutMs: number;

  constructor(limit: number, timeoutMs: number) {
    this.limit = limit;
    this.timeoutMs = timeoutMs;
  }

  /** 入队一个任务。队列满立即拒绝；排队超时也会拒绝（由 HTTP 层转成错误响应）。 */
  push(run: () => Promise<void>): Promise<void> {
    if (this.jobs.length >= this.limit) {
      return Promise.reject(new QueueFullError(`排队上限 ${this.limit}，请稍后再试`));
    }

    return new Promise<void>((resolve, reject) => {
      let timer: NodeJS.Timeout | undefined;

      const job: Job = {
        run: async () => {
          if (timer) clearTimeout(timer); // 开始执行：不再受排队超时约束
          try {
            await run();
            resolve();
          } catch (error) {
            reject(error instanceof Error ? error : new Error(String(error)));
          }
        },
      };

      timer = setTimeout(() => {
        const index = this.jobs.indexOf(job);
        if (index !== -1) this.jobs.splice(index, 1);
        reject(new Error(`排队超过 ${this.timeoutMs}ms，已放弃本次请求`));
      }, this.timeoutMs);

      this.jobs.push(job);
    });
  }

  /** 每次入队后调用一次；保证同一时刻只有一条 drain 循环在跑。 */
  kick(): void {
    if (this.draining) return;
    this.draining = true;
    void (async () => {
      try {
        while (this.jobs.length > 0) {
          const job = this.jobs.shift();
          if (job) await job.run();
        }
      } finally {
        this.draining = false;
      }
    })();
  }
}
排队和 steer / followUp 是两种语义，别混用： 串行队列把消息排成“下一条独立任务”，适合工单、指令类场景；session.steer() 把插话注入当前 run（第 09 课 9.1），适合聊天里“补充一句”。 按产品语义选，不是按实现难度选——这也是设置里 steeringMode / followUpMode 存在的原因。

生产篇 03 · 会话持久化

inMemory 换成按用户分目录的会话文件

高级篇 I.2 给了两条持久化路：① 会话文件按用户分目录（省事，pi 原生格式）；② 条目树进自己的数据库（可控，可加密可迁移）。这一节把①做成代码——核心是两个经过 verify 脚本核对的 API：SessionManager.create(cwd, dir) 建新会话，SessionManager.open(path) 打开已有会话续聊。

src/sessions-persistent.ts · 持久化会话注册表（完整代码）
import { mkdirSync, readdirSync, statSync } from "node:fs";
import { join } from "node:path";
import {
  createAgentSession,
  DefaultResourceLoader,
  getAgentDir,
  ModelRuntime,
  SessionManager,
  type AgentSession,
} from "@earendil-works/pi-coding-agent";
import { querySalesTool } from "./tools/query-sales.ts";
import { createTicketToolFor } from "./tools/create-ticket.ts";
import { createApprovalGate, type PendingApproval } from "./approvals.ts";

export type UserEntry = {
  session: AgentSession;
  running: boolean;
};

const registry = new Map<string, Promise<UserEntry>>();
const sessionsRoot = join(process.cwd(), "data", "sessions");

// userId 已过认证，但落盘路径仍然只信白名单字符——纵深防御不嫌多
function userSessionDir(userId: string): string {
  const dir = join(sessionsRoot, userId.replace(/[^a-zA-Z0-9_-]/g, "_"));
  mkdirSync(dir, { recursive: true });
  return dir;
}

/** 新用户建新会话；老用户打开目录里最新的会话文件继续聊。 */
function sessionManagerFor(userId: string): SessionManager {
  const dir = userSessionDir(userId);
  const latest = readdirSync(dir)
    .filter((name) => name.endsWith(".jsonl"))
    .map((name) => ({ name, mtimeMs: statSync(join(dir, name)).mtimeMs }))
    .sort((a, b) => b.mtimeMs - a.mtimeMs)[0];

  return latest
    ? SessionManager.open(join(dir, latest.name))
    : SessionManager.create(process.cwd(), dir);
}

export async function initModelRuntime(): Promise<ModelRuntime> {
  const modelRuntime = await ModelRuntime.create();
  if ((await modelRuntime.getAvailable()).length === 0) {
    throw new Error("没有可用模型：请先配置 Provider 和 API Key");
  }
  return modelRuntime;
}

// 审批卡要推给 SSE：宿主在 server.ts 里注入通知函数，避免模块循环依赖
let approvalNotifier: (approval: PendingApproval) => void = () => {};
export function setApprovalNotifier(fn: (approval: PendingApproval) => void): void {
  approvalNotifier = fn;
}

function createUserSession(userId: string, modelRuntime: ModelRuntime): Promise<UserEntry> {
  const loader = new DefaultResourceLoader({
    cwd: process.cwd(),
    agentDir: getAgentDir(),
    extensionFactories: [
      createApprovalGate({ userId, onRequest: (a) => approvalNotifier(a) }),
    ],
    systemPromptOverride: () =>
      [
        `你是用户 ${userId} 的销售与工单助手。`,
        "查数字前必须先使用 query_sales；创建工单用 create_ticket。",
        "不能编造数据；查询没有结果就明确说明。",
      ].join("\n"),
    appendSystemPromptOverride: () => [],
  });

  return (async () => {
    await loader.reload();
    const provider = process.env.AGENT_PROVIDER;
    const modelId = process.env.AGENT_MODEL;
    const model =
      (provider && modelId ? modelRuntime.getModel(provider, modelId) : undefined)
      ?? (await modelRuntime.getAvailable())[0];
    if (!model) throw new Error("请先配置一个可用模型");

    const { session } = await createAgentSession({
      model,
      modelRuntime,
      resourceLoader: loader,
      customTools: [querySalesTool, createTicketToolFor(userId)],
      tools: ["query_sales", "create_ticket"],
      sessionManager: sessionManagerFor(userId), // 6.3：持久化替换 inMemory
    });

    const entry: UserEntry = { session, running: false };
    session.subscribe((event) => {
      if (event.type === "agent_start") entry.running = true;
      if (event.type === "agent_settled") entry.running = false;
    });
    return entry;
  })();
}

/** 懒创建 + 去重：同一用户的并发首请求不会建两份会话 */
export function getOrCreate(userId: string, modelRuntime: ModelRuntime): Promise<UserEntry> {
  let entry = registry.get(userId);
  if (!entry) {
    entry = createUserSession(userId, modelRuntime);
    registry.set(userId, entry);
  }
  return entry;
}

/** 停机流程用：等所有会话空闲后再 dispose。 */
export async function allEntries(): Promise<UserEntry[]> {
  return Promise.all([...registry.values()]);
}
会话文件是明文 JSONL： 里面有用户输入、工具结果，可能还有敏感原文。三个后果：挂容器时要给 data/ 挂卷（本册 06）；合规要求高就走路②自建存储并加密；导出或分享会话前先审查（进阶篇 F.5 的清单）。官方替代入口是 SessionManager.list(cwd, dir) 与 findById，本册用纯 Node 读写目录是为了让“恢复逻辑”一眼可见。

生产篇 04 · 写操作审批流

挂起、推卡、恢复：企业 Agent 的标配回路

读操作可以放行，写操作必须有人点头。pi 的机制基础第 07 课已经铺好：tool_call 钩子返回 block: true 就拦截。这一节只补一个关键认识——钩子是异步的，可以挂起等一个“人的决定”：创建审批单、把卡片推到前端，等用户点批准或拒绝，再决定放行还是 block。超时默认拒绝（进阶篇 B.9 规则 3：安全判断做不出来就拒绝）。

src/approvals.ts · 审批门控（完整代码）
import { randomUUID } from "node:crypto";
import type { ExtensionFactory } from "@earendil-works/pi-coding-agent";

export type PendingApproval = {
  id: string;
  userId: string;
  toolName: string;
  argsSummary: string;
  resolve: (approved: boolean) => void;
};

const pending = new Map<string, PendingApproval>();

/** 工厂返回的扩展只挂在“这个用户”的会话上，所以 userId 用闭包注入。 */
export function createApprovalGate(options: {
  userId: string;
  timeoutMs?: number;
  onRequest?: (approval: PendingApproval) => void;
}): ExtensionFactory {
  const timeoutMs = options.timeoutMs ?? 120_000;

  return (pi) => {
    pi.on("tool_call", (event) => {
      // 只拦写操作；只读工具直接放行
      if (event.toolName !== "create_ticket") return;

      return new Promise((resolve) => {
        const finish = (approved: boolean) => {
          clearTimeout(timer);
          pending.delete(approval.id);
          resolve(
            approved
              ? undefined // 放行：继续执行 execute
              : { block: true, reason: "审批人拒绝了这次操作" },
          );
        };

        const approval: PendingApproval = {
          id: randomUUID(),
          userId: options.userId,
          toolName: event.toolName,
          argsSummary: JSON.stringify(event.input),
          resolve: finish,
        };

        const timer = setTimeout(() => finish(false), timeoutMs); // 超时 = 拒绝
        pending.set(approval.id, approval);
        options.onRequest?.(approval); // 宿主把审批卡推到 SSE
      });
    });
  };
}

/** 审批接口回调：只有本人能批本人的操作。 */
export function resolveApproval(id: string, userId: string, approved: boolean): boolean {
  const approval = pending.get(id);
  if (!approval || approval.userId !== userId) return false;
  approval.resolve(approved);
  return true;
}

完整的调用回路（时序）：

审批回路时序
用户: "帮我创建一个工单，标题：换键盘"
  → 模型发起 tool_call create_ticket
  → 审批门控挂起（pending 表 + 120s 超时）
  → SSE 推送 {"type":"approval_required","id":"...","toolName":"create_ticket","args":"..."}
  → 用户在另一个终端/页面 POST /approvals/<id> {"approve":true}
  → 门控 resolve(未 block) → execute() 落库
  → SSE 推送 tool_start / tool_end / final
超时 120s 未审批 → 自动 block，模型收到拒绝原因并向用户解释
审批是权限的一部分，不是豁免： 用户点了“批准”，不代表工具可以越权。工具的 execute 里仍然按 userId 过滤（6.3 的行级隔离）；审批通过的是“这次操作可以做”，授权决定的仍然是“这个人能对哪些数据做”。两道门各管各的。

生产篇 05 · 观测落地

trace 落盘、usage 记账、span 上 OTel

高级篇 H 搭好了三层观测面的骨架，这一节把它接进多用户服务：trace 按 userId 落 JSONL（排障用）、usage 从 message_end 落账（成本用）、run 级 span 用 OTLP/HTTP JSON 直接推给 Collector（接入公司 APM 用）。依旧零依赖。

src/observe.ts · 观测模块（完整代码）
import { appendFileSync, mkdirSync } from "node:fs";
import { join } from "node:path";
import { randomUUID } from "node:crypto";
import type { AgentSessionEvent } from "@earendil-works/pi-coding-agent";

const dataRoot = join(process.cwd(), "data");

function userFile(kind: string, userId: string): string {
  const dir = join(dataRoot, kind);
  mkdirSync(dir, { recursive: true });
  return join(dir, `${userId.replace(/[^a-zA-Z0-9_-]/g, "_")}.jsonl`);
}

function line(file: string, value: unknown): void {
  appendFileSync(file, JSON.stringify(value) + "\n");
}

/** trace + usage 订阅器：给每个用户会话挂一份（server.ts 里每次 run 挂上、收尾摘掉）。 */
export function createObserver(userId: string): (event: AgentSessionEvent) => void {
  const traceFile = userFile("trace", userId);
  const usageFile = userFile("usage", userId);

  const pendingTools = new Map<string, { toolName: string; t0: number }>();
  let runT0 = 0;

  return (event: AgentSessionEvent) => {
    const now = Date.now();
    switch (event.type) {
      case "agent_start":
        runT0 = now;
        line(traceFile, { kind: "run", phase: "start", ts: now });
        break;

      case "message_end": {
        // usage 的权威版本只在 message_end 的消息上（高级篇 K.4）
        if (event.message.role === "assistant" && event.message.usage) {
          const u = event.message.usage;
          line(usageFile, {
            ts: now,
            input: u.input ?? 0,
            output: u.output ?? 0,
            cacheRead: u.cacheRead ?? 0,
            costTotal: u.cost?.total ?? 0,
          });
        }
        break;
      }

      case "tool_execution_start":
        pendingTools.set(event.toolCallId, { toolName: event.toolName, t0: now });
        break;

      case "tool_execution_end": {
        const started = pendingTools.get(event.toolCallId);
        pendingTools.delete(event.toolCallId);
        // 被拦截的调用没有 durationMs：它没真正执行（第 07 课的时序提醒）
        line(traceFile, {
          kind: "tool",
          toolName: started?.toolName,
          durationMs: event.durationMs,
          isError: event.isError,
          ts: now,
        });
        break;
      }

      case "agent_settled":
        // 用户视角的“做完”：含重试与排队。算总时长不用 agent_end
        line(traceFile, {
          kind: "run",
          phase: "end",
          durationMs: now - runT0,
          aborted: event.aborted,
          ts: now,
        });
        otlpExport(userId, {
          name: "agent.run",
          t0: runT0,
          t1: now,
          attrs: { "user.id": userId, "run.aborted": event.aborted },
        });
        break;
    }
  };
}

// —— OTLP/HTTP JSON 零依赖导出：观测不拖垮业务，失败静默 ——

type Span = {
  name: string;
  t0: number;
  t1: number;
  attrs: Record<string, string | number | boolean>;
};

const otlpEndpoint = process.env.OTEL_EXPORTER_OTLP_ENDPOINT; // 如 http://localhost:4318

function otlpAttr(value: string | number | boolean) {
  if (typeof value === "string") return { value: { stringValue: value } };
  if (typeof value === "boolean") return { value: { boolValue: value } };
  return { value: { intValue: String(value) } };
}

function otlpExport(resource: string, span: Span): void {
  if (!otlpEndpoint) return;
  const nanos = (ms: number) => (BigInt(ms) * 1_000_000n).toString();
  void fetch(`${otlpEndpoint}/v1/traces`, {
    method: "POST",
    headers: { "content-type": "application/json" },
    body: JSON.stringify({
      resourceSpans: [
        {
          resource: {
            attributes: [{ key: "service.name", value: { stringValue: "data-agent" } }],
          },
          scopeSpans: [
            {
              scope: { name: resource },
              spans: [
                {
                  // 教学简化：每个 run 独立 trace。生产用 W3C Trace Context
                  // 把 HTTP 请求的 traceparent 传进 agent span，串成一条链
                  traceId: randomUUID().replaceAll("-", "").slice(0, 32),
                  spanId: randomUUID().replaceAll("-", "").slice(0, 16),
                  name: span.name,
                  kind: "SPAN_KIND_INTERNAL",
                  startTimeUnixNano: nanos(span.t0),
                  endTimeUnixNano: nanos(span.t1),
                  attributes: Object.entries(span.attrs).map(([key, value]) => ({
                    key,
                    ...otlpAttr(value),
                  })),
                },
              ],
            },
          ],
        },
      ],
    }),
  }).catch(() => {});
}

生产篇 06 · 容器与停机

打包你的 server，不是 pi CLI

进阶篇 F 容器化的是 pi 命令行工具；这一册容器化的是你写的服务。区别在于：镜像里跑的是 src/server.ts，要暴露端口、挂数据卷、给探针一个 /healthz，并且收到 SIGTERM 时优雅停机——四件事缺一件，容器编排都会给你“惊喜”。

src/server.ts · 最终拼装（完整代码）
import { createServer, type IncomingMessage, type ServerResponse } from "node:http";
import type { AgentSessionEvent } from "@earendil-works/pi-coding-agent";
import { authenticate, AuthError, type AuthUser } from "./auth.ts";
import { UserQueue } from "./queue.ts";
import {
  getOrCreate,
  initModelRuntime,
  setApprovalNotifier,
  allEntries,
} from "./sessions-persistent.ts";
import { resolveApproval } from "./approvals.ts";
import { createObserver } from "./observe.ts";

function sseWrite(res: ServerResponse, payload: unknown): void {
  res.write(`data: ${JSON.stringify(payload)}\n\n`);
}

function readJson(req: IncomingMessage): Promise<{ message?: string; approve?: boolean }> {
  return new Promise((resolve, reject) => {
    let data = "";
    req.on("data", (chunk: Buffer) => {
      data += chunk;
      if (data.length > 100_000) reject(new Error("body too large"));
    });
    req.on("end", () => {
      try { resolve(JSON.parse(data || "{}")); } catch { reject(new Error("bad json")); }
    });
    req.on("error", reject);
  });
}

async function main(): Promise<void> {
  const modelRuntime = await initModelRuntime();
  const streams = new Map<string, Set<ServerResponse>>(); // userId → 活跃 SSE 连接
  const queues = new Map<string, UserQueue>();

  // 审批卡从这里推到前端（sessions 模块回调进来，避免循环依赖）
  setApprovalNotifier((approval) => {
    for (const res of streams.get(approval.userId) ?? []) {
      sseWrite(res, {
        type: "approval_required",
        id: approval.id,
        toolName: approval.toolName,
        args: approval.argsSummary,
      });
    }
  });

  const server = createServer(async (req, res) => {
    const url = req.url ?? "";
    const method = req.method ?? "";

    // ① 健康检查：给容器探针用，不走认证
    if (method === "GET" && url === "/healthz") {
      res.writeHead(200).end("ok");
      return;
    }

    // ② 真实认证：替代 x-user-id 占位
    let user: AuthUser;
    try {
      user = authenticate(req);
    } catch (error) {
      const status = error instanceof AuthError ? error.status : 401;
      res.writeHead(status).end(JSON.stringify({ error: "unauthorized" }));
      return;
    }

    // ③ 审批端点：只有本人能批本人的操作
    if (method === "POST" && url.startsWith("/approvals/")) {
      const body = await readJson(req);
      const ok = resolveApproval(url.slice("/approvals/".length), user.userId, body.approve === true);
      res.writeHead(ok ? 200 : 404).end(JSON.stringify({ ok }));
      return;
    }

    if (method !== "POST" || url !== "/chat") {
      res.writeHead(404).end();
      return;
    }

    // ④ 输入校验
    const body = await readJson(req);
    const message = body.message?.trim() ?? "";
    if (message.length === 0 || message.length > 4000) {
      res.writeHead(400).end("invalid message");
      return;
    }

    // ⑤ 懒建会话（并发安全，实战篇 01）
    const entry = await getOrCreate(user.userId, modelRuntime);

    // ⑥ SSE 响应头
    res.writeHead(200, {
      "content-type": "text/event-stream",
      "cache-control": "no-cache",
      connection: "keep-alive",
    });

    // ⑦ 白名单事件流（高级篇 K：delta 渲染，end 落库）
    let closed = false;
    const unsubscribe = entry.session.subscribe((event: AgentSessionEvent) => {
      if (closed) return;
      switch (event.type) {
        case "message_update":
          if (event.assistantMessageEvent.type === "text_delta") {
            sseWrite(res, { type: "delta", delta: event.assistantMessageEvent.delta });
          }
          break;
        case "tool_execution_start":
          sseWrite(res, { type: "tool_start", toolName: event.toolName });
          break;
        case "tool_execution_end":
          sseWrite(res, { type: "tool_end", isError: event.isError });
          break;
      }
    });
    if (!streams.has(user.userId)) streams.set(user.userId, new Set());
    streams.get(user.userId)!.add(res);

    // ⑧ 断连：停止订阅 + 取消当前运行
    res.on("close", () => {
      if (closed) return;
      closed = true;
      unsubscribe();
      streams.get(user.userId)?.delete(res);
      if (entry.running) void entry.session.abort();
    });

    // ⑨ 忙时排队（6.2）：不再 409，有上限有超时
    const queue = queues.get(user.userId) ?? new UserQueue(5, 30_000);
    queues.set(user.userId, queue);

    try {
      await queue.push(async () => {
        if (closed) return; // 排到队首时客户端早走了
        const observer = createObserver(user.userId); // 6.5：trace + usage
        const unsubscribeObserver = entry.session.subscribe(observer);
        try {
          entry.running = true; // 同步占坑（实战篇修过的竞态）
          await entry.session.prompt(message);
          if (!closed) sseWrite(res, { type: "final", text: entry.session.getLastAssistantText() ?? "" });
        } catch {
          entry.running = false; // prompt 没进 run 就失败时，事件不会兜底，这里手动放行
          if (!closed) sseWrite(res, { type: "error", message: "服务暂时不可用，请稍后重试" });
        } finally {
          unsubscribeObserver();
        }
      });
    } catch (error) {
      // 队列满 / 排队超时：把明确的原因告诉客户端
      if (!closed) {
        sseWrite(res, { type: "error", message: error instanceof Error ? error.message : "请稍后重试" });
      }
    } finally {
      unsubscribe();
      if (!closed) res.end();
    }
  });

  // ⑩ 优雅停机：SIGTERM → 停接新请求 → 等所有会话空闲 → 清理 → 退出
  process.on("SIGTERM", () => {
    console.log("SIGTERM：停止接新请求，等在跑的任务结束……");
    server.close(async () => {
      const entries = await allEntries();
      await Promise.all(entries.map((e) => e.session.waitForIdle())); // agent_settled 语义
      for (const e of entries) e.session.dispose();
      await modelRuntime.settleCredentialWrites(); // 别丢刚刷新的 OAuth 令牌
      console.log("已优雅退出");
      process.exit(0);
    });
    setTimeout(() => process.exit(1), 15_000).unref(); // 兜底：之后编排层会 SIGKILL
  });

  server.listen(8787, () => console.log("listening on http://localhost:8787"));
}

main().catch((error) => {
  console.error(error);
  process.exit(1);
});

生产篇 07 · 演练与验收

亲手制造四个故障，然后才配说“能上线”

演练 · 四个必须亲手制造的故障
  1. 审批超时：让模型建工单，不批准，等 120 秒。观察模型收到“审批超时，已自动拒绝”并向用户解释——fail-closed 的用户体验。
  2. 断连烧钱：任务运行中关掉 curl/浏览器。服务端 trace 里 run 结束且 aborted: true，usage 里没有继续增长的 output。
  3. 发布重启：docker kill --signal=SIGTERM 一个正在跑任务的容器。观察日志顺序：SIGTERM → 等待 → 已优雅退出；重启后 alice 继续聊，历史还在（6.3 的恢复路径）。
  4. 越权审批：用 bob 的令牌 POST alice 的审批单。返回 404——resolveApproval 的本人校验在拦。

检查点：每个故障分别验证了哪一层？提示词在哪一层都没有出现——这就是“不依赖提示词的边界”。
